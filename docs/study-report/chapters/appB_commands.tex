\chapter{Command Reference}
\label{app:commands}

All commands run from the repository root. Everything that is a project command goes through \code{uv run}.

\section{Setup}

\begin{longtable}{@{}>{\raggedright\arraybackslash}p{7.2cm}>{\raggedright\arraybackslash}p{8.6cm}@{}}
\toprule
\textbf{Command} & \textbf{Purpose}\\
\midrule
\endhead
\code{uv sync} & Install runtime and development dependencies from \code{uv.lock}\\
\code{uv sync --frozen} & Install exactly the locked versions, never updating the lockfile\\
\code{uv sync --group eval} & Also install RAGAS and the evaluation dependencies\\
\code{cp .env.example .env} & Create the local environment file, then add \code{GOOGLE_API_KEY}\\
\code{uv run pre-commit install} & Install gitleaks, ruff, black and mypy hooks\\
\code{make install} & \code{uv sync} plus hook installation\\
\bottomrule
\end{longtable}

\section{The \code{vidyarag} command-line tool}

\begin{longtable}{@{}>{\raggedright\arraybackslash}p{7.2cm}>{\raggedright\arraybackslash}p{8.6cm}@{}}
\toprule
\textbf{Command} & \textbf{Purpose}\\
\midrule
\endhead
\code{vidyarag version} & Print the package version\\
\code{vidyarag config [-p PROFILE]} & Show the merged configuration of a profile\\
\code{vidyarag health} & Check settings, profile, Qdrant reachability and point count\\
\code{vidyarag download [-b SLUG] [--force]} & Download the PDFs and write the manifest\\
\code{vidyarag ingest [-p PROFILE] [--recreate] [--batch-size 64]} & Build or resume the index\\
\code{vidyarag ask "QUESTION" [-p PROFILE] [--context]} & Answer one question, optionally printing passages\\
\code{vidyarag eval -p PROFILE [--limit N] [--concurrency 3] [--rate 6.0] [--no-cache] [--compare baseline] [--goldset PATH]} & Run an evaluation; \code{--limit} writes to the scratch folder\\
\code{vidyarag report [PROFILES...]} & Compare the latest committed runs\\
\code{vidyarag goldset draft [--factual 28] [--multi-hop 20] [--unanswerable 12] [--seed 20260813] [--oversample 6] [--out PATH]} & Draft candidate questions and a review sheet\\
\code{vidyarag goldset check [--path PATH]} & Validate a gold set and show its composition\\
\code{vidyarag goldset unanswerable [--wanted 12] [--attempts 40] [--seed 20260816] [--write]} & Propose and verify unanswerable questions\\
\code{vidyarag goldset triage [--drafts PATH]} & Flag questions whose gold passage does not support the reference\\
\bottomrule
\end{longtable}

\section{Serving}

\begin{longtable}{@{}>{\raggedright\arraybackslash}p{7.2cm}>{\raggedright\arraybackslash}p{8.6cm}@{}}
\toprule
\textbf{Command} & \textbf{Purpose}\\
\midrule
\endhead
\code{uv run uvicorn vidyarag.api.main:app} & Start the HTTP API on port 8000 (\code{make serve} adds \code{--reload})\\
\code{uv run python app/app.py} & Start the Gradio demo on port 7860 (\code{make ui})\\
\code{curl http://localhost:8000/v1/health} & Health and point count\\
\code{curl http://localhost:8000/v1/config} & Active configuration\\
\code{curl -X POST http://localhost:8000/v1/query -H "Content-Type: application/json" -d '{"question": "..."}'} & Ask through the API\\
\code{curl -X POST http://localhost:8000/v1/search -H "Content-Type: application/json" -d '{"query": "...", "top_k": 5}'} & Retrieval only\\
\bottomrule
\end{longtable}

\section{Quality and tests}

\begin{longtable}{@{}>{\raggedright\arraybackslash}p{7.2cm}>{\raggedright\arraybackslash}p{8.6cm}@{}}
\toprule
\textbf{Command} & \textbf{Purpose}\\
\midrule
\endhead
\code{uv run pytest -m "not costly"} & Hermetic test suite (\code{make test})\\
\code{uv run pytest -m "not costly" --cov --cov-report=term-missing} & With coverage (\code{make test-cov})\\
\code{uv run pytest -m costly} & The three live tests; need a key and the index\\
\code{uv run ruff check .} and \code{uv run black --check .} & Lint and format check (\code{make lint})\\
\code{uv run ruff check --fix .} and \code{uv run black .} & Apply fixes (\code{make format})\\
\code{uv run mypy} & Strict type check (\code{make typecheck})\\
\code{make check} & Lint, type check and tests together\\
\bottomrule
\end{longtable}

\section{Containers}

\begin{longtable}{@{}>{\raggedright\arraybackslash}p{7.2cm}>{\raggedright\arraybackslash}p{8.6cm}@{}}
\toprule
\textbf{Command} & \textbf{Purpose}\\
\midrule
\endhead
\code{docker build -t vidyarag .} & Build the API image\\
\code{docker run -p 8000:8000 -v "$(pwd)/data/index:/app/data/index" -e GOOGLE_API_KEY=KEY vidyarag} & Run it with the index mounted writable\\
\code{docker compose up -d} and \code{docker compose down} & Start and stop a local Qdrant server (\code{make qdrant-up}, \code{make qdrant-down})\\
\code{QDRANT_MODE=server QDRANT_URL=http://localhost:6333 uv run vidyarag health} & Use the server instead of the embedded index\\
\bottomrule
\end{longtable}
