\section{Commands}

\begin{lstlisting}[language=bash]
# setup
uv sync                                  # runtime + dev, from uv.lock
uv sync --group eval                     # + RAGAS, only for evaluation
cp .env.example .env                     # paste a real GOOGLE_API_KEY
uv run pre-commit install                # gitleaks, hygiene, ruff, black, mypy

# data
uv run vidyarag download [-b biology] [--force]
uv run vidyarag ingest [--recreate] [--batch-size 64]
uv run vidyarag health                   # settings, profile, Qdrant, point count
uv run vidyarag config [-p guarded]      # merged profile, incl. thresholds + guards

# ask and serve
uv run vidyarag ask "QUESTION" [-p PROFILE] [--context]
uv run uvicorn vidyarag.api.main:app     # :8000, /docs
uv run python app/app.py                 # Gradio, :7860

# evaluate
uv run vidyarag eval -p guarded [--limit 4] [--rate 6] [--no-cache]
uv run vidyarag rejudge [PROFILES] [--rate 5] [--dry-run]
uv run vidyarag report [PROFILES]
uv run vidyarag goldset check|draft|unanswerable|triage

# quality
uv run pytest -m "not costly" [--cov]    # 466 hermetic tests
uv run pytest -m costly                  # 3 live tests (key + index)
uv run ruff check . && uv run black --check . && uv run mypy

# containers
docker build -t vidyarag .
docker run -p 8000:8000 -v "$(pwd)/data/index:/app/data/index" -e GOOGLE_API_KEY=... vidyarag
docker compose up -d                     # local Qdrant server on :6333

# the study report itself
cd docs/study-report && tectonic -X compile --outdir build main.tex
\end{lstlisting}
