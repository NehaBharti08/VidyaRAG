% --- Inline code and evidence references ---------------------------------------
% \code{...} and \ev{...} take their argument verbatim-like: raw underscores are
% allowed, "\_" is treated as "_", and line breaks are permitted after "_", "/"
% and "." so long identifiers and paths never overflow a table cell.
\ExplSyntaxOn
\cs_generate_variant:Nn \tl_replace_all:Nnn { NVn , NVV }
\tl_const:Nx \c_vr_us_tl { \tl_to_str:n { _ } }
\tl_const:Nx \c_vr_escus_tl { \c_backslash_str \tl_to_str:n { _ } }
\tl_new:N \l_vr_code_tl
\cs_new_protected:Npn \vr_breakable_string:n #1
  {
    \tl_set:Nx \l_vr_code_tl { \tl_to_str:n {#1} }
    \tl_replace_all:NVV \l_vr_code_tl \c_vr_escus_tl \c_vr_us_tl
    \tl_replace_all:NVn \l_vr_code_tl \c_vr_us_tl { \textunderscore \allowbreak }
    \tl_replace_all:Nnn \l_vr_code_tl { / } { / \allowbreak }
    \tl_replace_all:Nnn \l_vr_code_tl { . } { . \allowbreak }
    \tl_replace_all:Nnn \l_vr_code_tl { - } { - \allowbreak }
    \tl_use:N \l_vr_code_tl
  }
\NewDocumentCommand \code { m } { \texttt { \vr_breakable_string:n {#1} } }
\NewDocumentCommand \ev { m }
  { { \small \textsf { \color { vrgray } [ \vr_breakable_string:n {#1} ] } } }
\ExplSyntaxOff
